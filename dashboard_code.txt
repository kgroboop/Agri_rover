@app.route("/")
def index():
    return render_template_string(HTML_PAGE)

@app.route("/api/logs")
def get_logs():
    return jsonify(mission_logs)

# Endpoint where the Brain node pushes data ONLY when it completes a sample cycle
@app.route("/api/update", methods=["POST"])
def receive_update():
    global cycle_counter, mission_logs
    data = request.json
    if data:
        cycle_counter += 1
        entry = {
            "cycle": cycle_counter,
            "timestamp": datetime.now().strftime("%H:%M:%S"),
            "soil": int(data.get("soil", 0)),
            "temp": round(float(data.get("temp", 0.0)), 1),
            "humidity": round(float(data.get("humidity", 0.0)), 1),
            "pumpState": bool(data.get("pumpState", False)),
            "image": data.get("image", "")  # Base64 snapshot sent directly from Brain node
        }
        mission_logs.insert(0, entry)
        if len(mission_logs) > 30:
            mission_logs.pop()
        return jsonify({"status": "success"}), 200
    return jsonify({"status": "error"}), 400

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
