    class RoverServer(port: Int) : NanoHTTPD(port) {
        private val missionLogs = JSONArray()
        private var cycleCounter = 0
        private val aiExecutor = Executors.newSingleThreadExecutor()

        override fun serve(session: IHTTPSession): Response {
            val uri = session.uri
            val method = session.method

            // 1. Existing ESP32-S3 Brain Node endpoint
            if (method == Method.POST && uri == "/api/update") {
                return try {
                    val files = HashMap<String, String>()
                    session.parseBody(files)
                    val postData = files["postData"] ?: "{}"
                    val json = JSONObject(postData)

                    cycleCounter++
                    val timeStr = SimpleDateFormat("HH:mm:ss", Locale.getDefault()).format(Date())

                    val entry = JSONObject().apply {
                        put("cycle", cycleCounter)
                        put("timestamp", timeStr)
                        put("soil", json.optInt("soil", 0))
                        put("temp", json.optDouble("temp", 0.0))
                        put("humidity", json.optDouble("humidity", 0.0))
                        put("pumpState", json.optBoolean("pumpState", false))
                        put("image", json.optString("image", ""))
                        put("aiStatus", "UNANALYZED")
                        put("aiDiagnosis", "Tap 'Classify Cycle' for instant offline check.")
                    }

                    synchronized(missionLogs) {
                        val updatedLogs = JSONArray()
                        updatedLogs.put(entry)
                        for (i in 0 until minOf(missionLogs.length(), 29)) {
                            updatedLogs.put(missionLogs.getJSONObject(i))
                        }
                        while (missionLogs.length() > 0) missionLogs.remove(0)
                        for (i in 0 until updatedLogs.length()) {
                            missionLogs.put(updatedLogs.getJSONObject(i))
                        }
                    }

                    newFixedLengthResponse(Response.Status.OK, "application/json", "{\"status\":\"success\"}")
                } catch (e: Exception) {
                    newFixedLengthResponse(Response.Status.BAD_REQUEST, "application/json", "{\"status\":\"error\"}")
                }
            }

            // 2. Dashboard UI polling endpoint
            if (method == Method.GET && uri == "/api/logs") {
                val snapshot = synchronized(missionLogs) { missionLogs.toString() }
                return newFixedLengthResponse(Response.Status.OK, "application/json", snapshot)
            }

            // 3. Per-cycle offline analysis endpoint
            if (method == Method.POST && uri == "/api/analyze") {
                return try {
                    val files = HashMap<String, String>()
                    session.parseBody(files)
                    val postData = files["postData"] ?: "{}"
                    val reqJson = JSONObject(postData)
                    val targetCycle = reqJson.optInt("cycle", -1)
                    val sensitivity = reqJson.optString("sensitivity", "medium")

                    var targetObj: JSONObject? = null
                    synchronized(missionLogs) {
                        for (i in 0 until missionLogs.length()) {
                            val item = missionLogs.getJSONObject(i)
                            if (targetCycle == -1 || item.optInt("cycle", -1) == targetCycle) {
                                targetObj = item
                                break
                            }
                        }
                    }

                    if (targetObj == null) {
                        newFixedLengthResponse(
                            Response.Status.NOT_FOUND,
                            "application/json",
                            "{\"ok\":false,\"error\":\"Log entry not found.\"}"
                        )
                    } else {
                        val logEntry = targetObj!!
                        val imageField = reqJson.optString("image", logEntry.optString("image", ""))

                        if (imageField.isBlank()) {
                            logEntry.put("aiStatus", "NOT A PLANT")
                            logEntry.put("aiDiagnosis", "NOT A PLANT\nNo snapshot image available in this cycle.")
                            newFixedLengthResponse(Response.Status.OK, "application/json", logEntry.toString())
                        } else {
                            val result = classifyLeafSimple(imageField, logEntry, sensitivity)
                            logEntry.put("aiStatus", result.first)
                            logEntry.put("aiDiagnosis", result.second)
                            newFixedLengthResponse(Response.Status.OK, "application/json", logEntry.toString())
                        }
                    }
                } catch (e: Exception) {
                    val errJson = JSONObject().apply {
                        put("ok", false)
                        put("error", e.message ?: "Analysis failed")
                    }
                    newFixedLengthResponse(Response.Status.INTERNAL_ERROR, "application/json", errJson.toString())
                }
            }

            // 4. Dashboard HTML UI
            return newFixedLengthResponse(Response.Status.OK, "text/html", DASHBOARD_HTML)
        }

        private fun classifyLeafSimple(
            imageData: String,
            telemetry: JSONObject,
            sensitivity: String
        ): Pair<String, String> {
            val future = aiExecutor.submit<Pair<String, String>> {
                val soil = telemetry.optInt("soil", 0)
                val temp = telemetry.optDouble("temp", 0.0)
                val humidity = telemetry.optDouble("humidity", 0.0)
                val pumpOn = telemetry.optBoolean("pumpState", false)
                val teleSummary = "Soil $soil%, Temp ${String.format(Locale.US, "%.1f", temp)}°C, Humidity ${String.format(Locale.US, "%.1f", humidity)}%, Pump ${if (pumpOn) "ON" else "OFF"}."

                try {
                    val cleanBase64 = if (imageData.contains(",")) {
                        imageData.substringAfter(",").trim()
                    } else {
                        imageData.trim()
                    }

                    val bytes = Base64.decode(cleanBase64, Base64.DEFAULT)
                    val decoded: Bitmap? = BitmapFactory.decodeByteArray(bytes, 0, bytes.size)
                    if (decoded == null) {
                        return@submit Pair(
                            "NOT A PLANT",
                            "NOT A PLANT\nImage could not be decoded; defaulting to NOT A PLANT.\n$teleSummary"
                        )
                    }

                    val w = 120
                    val h = 120
                    val scaled = Bitmap.createScaledBitmap(decoded, w, h, true)
                    if (scaled != decoded) decoded.recycle()

                    val xStart = (w * 0.20).toInt()
                    val xEnd = (w * 0.80).toInt()
                    val yStart = (h * 0.20).toInt()
                    val yEnd = (h * 0.80).toInt()

                    var validPixels = 0
                    var greenPixels = 0
                    var lesionPixels = 0

                    val hsv = FloatArray(3)
                    for (y in yStart until yEnd) {
                        for (x in xStart until xEnd) {
                            val px = scaled.getPixel(x, y)
                            val r = Color.red(px)
                            val g = Color.green(px)
                            val b = Color.blue(px)

                            val maxC = max(r, max(g, b))
                            val minC = min(r, min(g, b))
                            if (maxC < 25 || (maxC > 245 && (maxC - minC) < 15)) continue

                            validPixels++
                            Color.RGBToHSV(r, g, b, hsv)
                            val hue = hsv[0]
                            val sat = hsv[1]
                            val valB = hsv[2]

                            // Green leaf tone
                            val isGreen = (hue in 68f..165f) && (sat >= 0.20f) && (valB >= 0.16f) && (g >= r + 6)
                            // Yellowing or brown/spotted leaf lesion tone
                            val isLesion = ((hue in 35f..67f) && sat >= 0.30f && valB >= 0.22f) ||
                                    ((hue in 5f..34f) && sat >= 0.30f && valB in 0.14f..0.75f && r >= g + 5)

                            if (isGreen) greenPixels++
                            if (isLesion) lesionPixels++
                        }
                    }
                    scaled.recycle()

                    val greenRatio = if (validPixels > 0) greenPixels.toFloat() / validPixels.toFloat() else 0f
                    val vegTotal = greenPixels + lesionPixels
                    val lesionShare = if (vegTotal > 0) lesionPixels.toFloat() / vegTotal.toFloat() else 0f

                    // Anything without clear plant presence is NOT A PLANT
                    if (validPixels == 0 || greenPixels < 60 || greenRatio < 0.10f) {
                        return@submit Pair(
                            "NOT A PLANT",
                            "NOT A PLANT\nNo clear plant foliage detected in center frame (${String.format(Locale.US, "%.0f", greenRatio * 100)}% green).\n$teleSummary"
                        )
                    }

                    val lesionCutoff = when (sensitivity.lowercase(Locale.getDefault())) {
                        "low" -> 0.20f
                        "high" -> 0.05f
                        else -> 0.12f
                    }

                    val label = if (lesionShare >= lesionCutoff) "DISEASED" else "HEALTHY"
                    val line1 = if (label == "DISEASED") {
                        "Leaf discoloration/spots detected (${String.format(Locale.US, "%.0f", lesionShare * 100)}% lesion share, ${String.format(Locale.US, "%.0f", greenRatio * 100)}% green)."
                    } else {
                        "Healthy green leaf appearance (${String.format(Locale.US, "%.0f", greenRatio * 100)}% green, ${String.format(Locale.US, "%.0f", lesionShare * 100)}% lesion share)."
                    }
                    val line2 = if (label == "DISEASED") {
                        "Inspect affected leaves and check moisture; $teleSummary"
                    } else {
                        "Foliage looks normal; $teleSummary"
                    }

                    Pair(label, "$label\n$line1\n$line2")
                } catch (e: Exception) {
                    Pair(
                        "NOT A PLANT",
                        "NOT A PLANT\nCould not process snapshot (${e.javaClass.simpleName}); defaulting to NOT A PLANT.\n$teleSummary"
                    )
                }
            }

            return future.get()
        }

                    <script>
                        var latestLogs = [];
                        var busyCycles = {};
                        var batchRunning = false;

                        function getSensitivity() {
                            var sel = document.getElementById('sensitivitySelect');
                            return sel ? sel.value : 'medium';
                        }

                        function badgeClassFor(status) {
                            if (status === 'HEALTHY') return 'ai-tag tag-healthy';
                            if (status === 'DISEASED') return 'ai-tag tag-diseased';
                            if (status === 'NOT A PLANT') return 'ai-tag tag-not-plant';
                            return 'ai-tag';
                        }

                        async function classifyCycle(cycleNum) {
                            var entry = latestLogs.find(function(x) { return x.cycle === cycleNum; });
                            if (!entry || !entry.image) return;

                            busyCycles[cycleNum] = true;
                            renderLogs();

                            try {
                                await fetch('/api/analyze', {
                                    method: 'POST',
                                    headers: { 'Content-Type': 'application/json' },
                                    body: JSON.stringify({
                                        cycle: cycleNum,
                                        image: entry.image,
                                        sensitivity: getSensitivity()
                                    })
                                });
                            } catch (err) {
                                console.error('Analysis error:', err);
                            } finally {
                                delete busyCycles[cycleNum];
                                await fetchLogs();
                            }
                        }

                        async function classifyAllPending() {
                            if (batchRunning) return;
                            batchRunning = true;
                            var batchBtn = document.getElementById('batchBtn');
                            if (batchBtn) batchBtn.disabled = true;

                            try {
                                var pending = latestLogs.filter(function(item) {
                                    return item.image && item.aiStatus === 'UNANALYZED';
                                });
                                for (var i = 0; i < pending.length; i++) {
                                    await classifyCycle(pending[i].cycle);
                                }
                            } finally {
                                batchRunning = false;
                                if (batchBtn) batchBtn.disabled = false;
                            }
                        }

                        function renderLogs() {
                            var container = document.getElementById('logContainer');
                            if (!latestLogs || latestLogs.length === 0) {
                                container.innerHTML = '<div class="empty-state">App Ready — Waiting for Rover Cycle...</div>';
                                return;
                            }

                            container.innerHTML = latestLogs.map(function(item) {
                                var isBusy = !!busyCycles[item.cycle];
                                var statusText = isBusy ? 'ANALYZING' : (item.aiStatus || 'UNANALYZED');
                                var noteText = isBusy ? 'Analyzing snapshot locally...' : (item.aiDiagnosis || 'Ready for check.');

                                return '<div class="cycle-card">' +
                                    '<div class="cycle-top">' +
                                        '<div class="cycle-title">Cycle #' + item.cycle + ' — ' + item.timestamp + '</div>' +
                                        '<span class="' + badgeClassFor(statusText) + '">' + statusText + '</span>' +
                                    '</div>' +
                                    '<div class="photo-box">' +
                                        '<img src="' + item.image + '" alt="Rover Sample Snapshot">' +
                                    '</div>' +
                                    '<div class="grid">' +
                                        '<div class="metric-box">' +
                                            '<div class="metric-label">SOIL</div>' +
                                            '<div class="metric-value">' + item.soil + '%</div>' +
                                        '</div>' +
                                        '<div class="metric-box">' +
                                            '<div class="metric-label">TEMP</div>' +
                                            '<div class="metric-value">' + Number(item.temp).toFixed(1) + ' °C</div>' +
                                        '</div>' +
                                        '<div class="metric-box">' +
                                            '<div class="metric-label">HUMIDITY</div>' +
                                            '<div class="metric-value">' + Number(item.humidity).toFixed(1) + ' %</div>' +
